
  // Copy Candidate Data for Teletalk
  const handleCopyCandidateData = (app: ApplicationRequest) => {
    const formattedData = "[TELETALK AUTO-FILL DATA]\n" +
      "Name: " + app.candidateName + "\n" +
      "Phone: " + app.candidatePhone + "\n" +
      "District: " + app.candidateDistrict + "\n" +
      "Job: " + app.jobTitle + "\n" +
      "Post: " + app.postName + "\n" +
      "Government Fee: ?" + (app.governmentFee || 200) + "\n" +
      "Tracking ID: " + app.id;
    navigator.clipboard.writeText(formattedData);
    setCopiedCandidateId(app.id);
    setTimeout(() => setCopiedCandidateId(null), 2500);
  };
